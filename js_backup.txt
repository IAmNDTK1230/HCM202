    function getRevealCount(answer, secondsLeft){
      const totalLetters = getLetterCount(answer);
      if(totalLetters <= 1) return 0;
      if(secondsLeft <= 0) return totalLetters;

      if(secondsLeft > 25) return 0;

      let count = 0;

      // 25s -> 20s: m??? ch???m, m???i m???c 5s kho???ng 1 ch???.
      count += Math.max(0, Math.floor((25 - secondsLeft) / 5) + 1);

      // T??? 15s tr??? xu???ng: t??ng nh???p, m???i m???c 5s m??? th??m 1-2 ch???.
      if(secondsLeft <= 15){
        const fastSteps = Math.floor((15 - secondsLeft) / 5) + 1;
        for(let i = 1; i <= fastSteps; i++){
          count += i % 2 === 0 ? 2 : 1;
        }
      }

      return Math.min(totalLetters, count);
    }

      const indexes = [];

      score = 0; round = 0; timeLeft = QUESTION_TIME;

        await updateDoc(roomRef(), {
